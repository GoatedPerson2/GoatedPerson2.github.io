-- server-checked punch, used in my combat system
local Hitbox = require(script.Hitbox)
function CombatService.Punch(player, side)
if not Sanity.check(player, "Punch") then return end
local hits = Hitbox.cast(player.Character, 5.5)
for _, target in hits do
Damage.apply(target, 12, side)
end
end
